cmake_minimum_required(VERSION 3.18)
project(spoofgpu CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(spoofgpu SHARED module.cpp)
target_include_directories(spoofgpu PRIVATE ${CMAKE_SOURCE_DIR})
target_compile_options(spoofgpu PRIVATE -fvisibility=hidden -Os)
option(SG_DEBUG "Log to logcat (tag SpoofGame)" OFF)
if(SG_DEBUG)
  target_compile_definitions(spoofgpu PRIVATE SG_DEBUG=1)
endif()
target_link_libraries(spoofgpu PRIVATE dl log)
set_target_properties(spoofgpu PROPERTIES PREFIX "" OUTPUT_NAME "arm64-v8a")
