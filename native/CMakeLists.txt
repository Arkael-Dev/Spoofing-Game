cmake_minimum_required(VERSION 3.18)
project(spoofgpu CXX)
set(CMAKE_CXX_STANDARD 17)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(Dobby ${CMAKE_BINARY_DIR}/dobby)
add_library(spoofgpu SHARED module.cpp)
target_include_directories(spoofgpu PRIVATE ${CMAKE_SOURCE_DIR} ${CMAKE_SOURCE_DIR}/Dobby/include)
target_compile_options(spoofgpu PRIVATE -fvisibility=hidden -Os)
target_link_libraries(spoofgpu PRIVATE dobby log)
set_target_properties(spoofgpu PROPERTIES PREFIX "" OUTPUT_NAME "arm64-v8a")
